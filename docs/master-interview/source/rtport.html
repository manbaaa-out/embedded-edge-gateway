Resume usage of the other clock. */ 714 portNVIC_SYSTICK_CTRL_REG = portNVIC_SYSTICK_CLK_BIT | portNVIC_SYSTICK_INT_BIT; 715 716 if( ( portNVIC_SYSTICK_CTRL_REG & portNVIC_SYSTICK_COUNT_FLAG_BIT ) != 0 ) 717 { 718 /* The partial tick period already ended. Be sure the SysTick 719 * counts it only once. */ 720 portNVIC_SYSTICK_CURRENT_VALUE_REG = 0; 721 } 722 723 portNVIC_SYSTICK_LOAD_REG = ulTimerCountsForOneTick - 1UL; 724 portNVIC_SYSTICK_CTRL_REG = portNVIC_SYSTICK_CLK_BIT_CONFIG | portNVIC_SYSTICK_INT_BIT | portNVIC_SYSTICK_ENABLE_BIT; 725 } 726 #endif /* portNVIC_SYSTICK_CLK_BIT_CONFIG */ 727 728 /* Step the tick to account for any tick periods that elapsed. */ 729 vTaskStepTick( ulCompleteTickPeriods ); 730 731 /* Exit with interrupts enabled. */ 732 __asm volatile ( "cpsie i" ::: "memory" );
